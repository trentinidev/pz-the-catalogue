module Catalogue
{
    imports { Base }

    /* The book on the ground: vanilla's closed-catalogue MESH wearing our own cover.

       A new mesh would mean shipping an FBX, and nothing about this item needs a shape
       vanilla does not already have -- it is a paperback-sized book lying shut. What it
       needed was to stop being a Keio Furniture catalogue.

       The texture layout is not ours to choose: it re-skins WorldItems/Catalogue, so the
       mesh's UVs decide it. Measured off vanilla's own 64x64, columns 0..56 are the cover
       and 57..63 are the fore-edge. tools/gen_art.ps1 builds it to that layout from the
       art renders; do not edit the PNG by hand. */
    model ShopCatalogueWorld
    {
        mesh = WorldItems/Catalogue,
        texture = TheCatalogue/ShopCatalogueWorld,
        scale = 0.125,
    }

    /* The catalogue itself. Deliberately a plain "normal" item and NOT literature:
       literature items get a Read context option and a page/boredom system we do not
       want competing with our own Buy/Sell entries on the right-click menu.

       The inventory icon and the ground model are ours. StaticModel -- the open book the
       character holds while placing an order -- is still vanilla's: it is a different
       mesh with a different texture layout, showing a two-page spread, and the cover art
       has no spread to put on it. */
    item Catalogue
    {
        DisplayCategory = Literature,
        ItemType = base:normal,
        Weight = 0.5,
        Icon = ShopCatalogue,
        StaticModel = Catalogue_Open,
        WorldStaticModel = ShopCatalogueWorld,
        Tags = base:isfirefuel;base:isfiretinder,
        FireFuelRatio = 0.25,
    }

    /* Oversized shipping parcels.

       Vanilla tops out at Parcel_ExtraLarge, capacity 20, and a single generator
       weighs 40 -- so the largest box the game ships cannot hold one item the
       catalogue sells. Rather than dropping heavy goods loose on the ground and
       calling it a delivery, the company packs them in bigger crates.

       THREE TIERS, THREE MATERIALS. They used to be one cardboard box drawn three
       times, which made the capacity the only thing telling them apart -- and capacity
       is a number in a tooltip, not something you can see on the ground. Now the step
       up in size is a step up in packaging: a taped carton, a strapped wooden crate, a
       tarped pallet load. That is legible at a glance and it is also just what a real
       freight company would do.

       Sound stays vanilla's largest parcel throughout, and weight stays at 1.0 like
       every vanilla parcel, so the box itself never becomes the reason a delivery is
       too heavy. */
    /* Three meshes of our own, built by tools/blender_parcels.py and editable in
       art/models/*.blend.

       These used to re-skin vanilla's parcel mesh, which meant the crate had no corner
       frame and the pallet load had no pallet: silhouette is shape, and no texture fakes
       it at the angle the game draws items from. It also meant the texture had to be a
       re-materialised copy of vanilla's atlas, because that atlas's UV layout lives
       inside a binary FBX -- paint a label onto it and you learn where it went by
       dropping a parcel and walking around it.

       Owning the mesh settles both. The frame and the pallet are real geometry, and the
       unwrap is a grid we chose and wrote down, so a pixel in the FRONT cell comes out on
       the front. See art/models/README.md and art/models/uv_layout.png.

       scale = 1.0 because these are modelled life-size: vanilla's remaining ASCII .x
       models measure like real objects at scale 1.0, so a metre here is a metre there. */
    model Parcel25World
    {
        mesh = WorldItems/parcel25,
        texture = TheCatalogue/Parcel25World,
        scale = 1.0,
    }

    model Parcel50World
    {
        mesh = WorldItems/parcel50,
        texture = TheCatalogue/Parcel50World,
        scale = 1.0,
    }

    model Parcel100World
    {
        mesh = WorldItems/parcel100,
        texture = TheCatalogue/Parcel100World,
        scale = 1.0,
    }

    item Parcel_XXL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = ParcelXXL,
        Capacity = 25,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel25World,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    item Parcel_5XL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = Parcel5XL,
        Capacity = 50,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel50World,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    item Parcel_10XL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = Parcel10XL,
        Capacity = 100,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel100World,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    /* Bundle loose notes. The exact mirror of vanilla's UnbundleMoney -- same time,
       same tags, same category, same 1:100 rate -- so the two read as one pair of
       recipes rather than as a mod bolted onto a vanilla one.

       Vanilla can only take a bundle apart, never put one together, which is an odd
       gap in a game that spawns thousands of loose notes: each one is a separate
       InventoryItem at 0.01 weight, so a few thousand dollars is a few thousand
       objects and tens of kilos. Bundling halves the weight and cuts the object count
       by a hundred, and until now the catalogue was the only thing in the game that
       could do it. */
    craftRecipe BundleMoney
    {
        timedAction = Making,
        time = 80,
        Tags = InHandCraft;Packing;CanBeDoneInDark,
        category = Packing,
        inputs
        {
            item 100 [Base.Money] flags[AllowFavorite;InheritFavorite],
        }
        outputs
        {
            item 1 Base.MoneyBundle,
        }
    }

    /* One Notebook in, one Catalogue out. The notebook is consumed.
       InHandCraft keeps it out of the workstation-only lists, and CanBeDoneInDark
       matches the vanilla paper recipes -- writing a shopping list needs no light. */
    craftRecipe MakeShopCatalogue
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;CanBeDoneInDark,
        category = Miscellaneous,
        inputs
        {
            item 1 [Base.Notebook],
        }
        outputs
        {
            item 1 Catalogue.Catalogue,
        }
    }
}

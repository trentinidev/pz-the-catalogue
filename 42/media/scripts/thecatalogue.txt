module Catalogue
{
    imports { Base }

    /* The catalogue itself. Deliberately a plain "normal" item and NOT literature:
       literature items get a Read context option and a page/boredom system we do not
       want competing with our own Buy/Sell entries on the right-click menu.
       Art is borrowed wholesale from vanilla Base.Catalog, which already ships an
       inventory icon, an open-book static model and a ground model. */
    item Catalogue
    {
        DisplayCategory = Literature,
        ItemType = base:normal,
        Weight = 0.5,
        Icon = ShopCatalogue,
        StaticModel = Catalogue_Open,
        WorldStaticModel = Catalogue,
        Tags = base:isfirefuel;base:isfiretinder,
        FireFuelRatio = 0.25,
    }

    /* Oversized shipping parcels.

       Vanilla tops out at Parcel_ExtraLarge, capacity 20, and a single generator
       weighs 40 -- so the largest box the game ships cannot hold one item the
       catalogue sells. Rather than dropping heavy goods loose on the ground and
       calling it a delivery, the company packs them in bigger crates.

       Art and sound are vanilla's largest parcel throughout: same Parcel1 icon, same
       world model, same open/close/put-in sounds. A crate that is visibly the same
       object at four times the capacity is a small fiction, and a much smaller one
       than a generator arriving with no packaging at all.

       Capacity is the only thing that differs. Weight stays at 1.0 like every vanilla
       parcel, so the box itself never becomes the reason a delivery is too heavy. */
    item Parcel_XXL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = Parcel1,
        Capacity = 25,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel1,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    item Parcel_5XL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = Parcel1,
        Capacity = 50,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel1,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    item Parcel_10XL
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 1.0,
        Icon = Parcel1,
        Capacity = 100,
        CloseSound = CloseParcelLarge,
        OpenSound = OpenParcelLarge,
        PutInSound = StoreItemParcelLarge,
        WorldStaticModel = Parcel1,
        Tags = base:neverempty;base:isfirefuel;base:isfiretinder,
    }

    /* Bundle loose notes. The exact mirror of vanilla's UnbundleMoney -- same time,
       same tags, same category, same 1:100 rate -- so the two read as one pair of
       recipes rather than as a mod bolted onto a vanilla one.

       Vanilla can only take a bundle apart, never put one together, which is an odd
       gap in a game that spawns thousands of loose notes: each one is a separate
       InventoryItem at 0.01 weight, so a few thousand dollars is a few thousand
       objects and tens of kilos. Bundling halves the weight and cuts the object count
       by a hundred, and until now the catalogue was the only thing in the game that
       could do it. */
    craftRecipe BundleMoney
    {
        timedAction = Making,
        time = 80,
        Tags = InHandCraft;Packing;CanBeDoneInDark,
        category = Packing,
        inputs
        {
            item 100 [Base.Money] flags[AllowFavorite;InheritFavorite],
        }
        outputs
        {
            item 1 Base.MoneyBundle,
        }
    }

    /* One Notebook in, one Catalogue out. The notebook is consumed.
       InHandCraft keeps it out of the workstation-only lists, and CanBeDoneInDark
       matches the vanilla paper recipes -- writing a shopping list needs no light. */
    craftRecipe MakeShopCatalogue
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;CanBeDoneInDark,
        category = Miscellaneous,
        inputs
        {
            item 1 [Base.Notebook],
        }
        outputs
        {
            item 1 Catalogue.Catalogue,
        }
    }
}

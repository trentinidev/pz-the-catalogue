/* The Catalogue -- sandbox options.

   FORMAT WARNINGS, both of which fail silently:
     * this file is parsed by ScriptParser, so comments are the C-style block form ONLY.
       A // line comment is not merely unsupported -- the parser has no rule for it, the
       text survives into the token stream as garbage, and the whole block silently fails
       to register with nothing written to the log.
     * every option id must contain EXACTLY ONE dot. SandboxOptions.parseName splits on it
       and falls back to a null table name when the split is not two parts, which drops the
       option into the root of SandboxVars instead of under TheCatalogue.

   WHERE TO FIND THESE IN GAME: custom sandbox options are drawn by the SERVER SETTINGS
   screen (Host -> Manage settings), not by the singleplayer sandbox screen -- the only
   consumer of SandboxOptions.isCustom() in the game's Lua is ServerSettingsScreen.lua.

   A SAVE ALREADY IN PROGRESS DOES NOT SEE A CHANGED DEFAULT. Every option a save knows
   about is written into it when it is created, and TC.opt reads the save before it
   reads our defaults -- deliberately, because a player's setting must beat ours. So
   changing `default =` here reaches new saves only.

   There is no <save>_SandboxVars.lua in B42; the values live in the save's map_sand.bin
   as Java writeUTF pairs. `sh tools/patch_save_sandbox.sh --apply` rewrites them in
   place, with the game closed and a backup taken.
*/

VERSION = 1,

/* Scales every buy price. Raise it to make the world expensive and money scarce,
   lower it to make looting cash meaningful. Sell prices follow automatically, since
   they are derived from the buy price. */
option TheCatalogue.PriceMultiplier
{
    type = double,
    min = 0.05, max = 20.0,
    default = 5.0,
    page = TheCatalogue,
    translation = TheCatalogue_PriceMultiplier,
}

/* Fraction of an item's value paid out when selling. 0.10 pays a tenth of what the
   catalogue charges. Setting this to 1.0 makes buy and sell lossless, which turns any
   craft that adds value into an infinite money loop -- deliberately allowed. */
option TheCatalogue.SellRatio
{
    type = double,
    min = 0.0, max = 1.0,
    default = 0.10,
    page = TheCatalogue,
    translation = TheCatalogue_SellRatio,
}

/* Upper bound on the quantity spinner in the buy panel. */
option TheCatalogue.MaxQuantityPerPurchase
{
    type = integer,
    min = 1, max = 1000,
    default = 100,
    page = TheCatalogue,
    translation = TheCatalogue_MaxQuantityPerPurchase,
}

/* When on, selling a bag or a loaded gun sells its contents too, at their own value.
   When off, the catalogue refuses non-empty containers and loaded firearms. */
option TheCatalogue.SellContainerContents
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_SellContainerContents,
}

/* How long deliveries take, as a multiplier on the calculated lead time. The lead
   time itself grows with the weight and value of the order -- a tin of beans is a few
   hours, a generator most of a day. Set to 0.01 for near-instant delivery. */
option TheCatalogue.DeliveryHoursMultiplier
{
    type = double,
    min = 0.01, max = 10.0,
    default = 1.0,
    page = TheCatalogue,
    translation = TheCatalogue_DeliveryHoursMultiplier,
}

/* Surcharge for skipping the wait, as a percentage of the order. */
option TheCatalogue.RushFeePercent
{
    type = integer,
    min = 0, max = 500,
    default = 20,
    page = TheCatalogue,
    translation = TheCatalogue_RushFeePercent,
}

/* Diagnostic logging. Off by default: the timing and loot-table lines are useful
   while working on the mod and are noise in a player console. Real problems --
   refused purchases, refunds, API misuse -- always print regardless. */
option TheCatalogue.DebugLogging
{
    type = boolean,
    default = false,
    page = TheCatalogue,
    translation = TheCatalogue_DebugLogging,
}

/* Seconds spent placing an order before it completes, as an interruptible action.
   0 makes buying instant, which is how the mod behaved before this existed. */
option TheCatalogue.OrderSeconds
{
    type = double,
    min = 0.0, max = 15.0,
    default = 2.0,
    page = TheCatalogue,
    translation = TheCatalogue_OrderSeconds,
}

/* How common the Shop Catalogue is in world loot, relative to the built-in weights.
   Set to 0 to remove it from loot entirely and leave crafting as the only route. */
option TheCatalogue.CatalogueLootMultiplier
{
    type = double,
    min = 0.0, max = 10.0,
    default = 1.0,
    page = TheCatalogue,
    translation = TheCatalogue_CatalogueLootMultiplier,
}

/* When on, both windows close if the player no longer has a Shop Catalogue on them.
   Turn it off for an arcade-style game where the catalogue is a menu rather than an
   object you have to be carrying. */
option TheCatalogue.RequireCatalogueOnPerson
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_RequireCatalogueOnPerson,
}

/* When on, the catalogue pays out of any container the character can reach -- the crate
   in the safehouse, the till on the counter, the car boot -- and not only out of their
   own pockets. The reach rule is the game's own: the containers your loot window is
   showing you are the containers the catalogue can see.

   Turn it off to make the weight of cash a real ceiling again: at about $5,000 per
   backpack, pockets-only is what limits a single purchase. */
option TheCatalogue.SpendFromContainers
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_SpendFromContainers,
}

/* When on, the company pays more for what it actually wants back: roughly 25% for
   weapons, ammunition, medicine, tools and radios, the ordinary rate for most things,
   and about 5% for ornaments, mementoes and improvised clubs. Multipliers on the sell
   ratio, so that dial still scales everything together.

   Off pays one rate for everything, which is how the mod behaved before 0.15.0. */
option TheCatalogue.CategorySellRates
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_CategorySellRates,
}

/* When OFF (the default) the catalogue refuses to buy food that rots -- fresh produce,
   meat, caught fish -- while still happily selling it to you. Tins, sacks and anything
   else that keeps are unaffected.

   The reason is that farming, fishing and foraging each produce dozens of items a trip,
   nearly all of them at the one-dollar floor, at no risk and without end. Turn it on if
   you would rather the catalogue bought your harvest. */
option TheCatalogue.AcceptPerishableFood
{
    type = boolean,
    default = false,
    page = TheCatalogue,
    translation = TheCatalogue_AcceptPerishableFood,
}

/* How many of the dearer items the company can supply in a game week. 1.0 is one
   generator, three pistols or ten crowbars; 2.0 doubles every band; 0 removes the
   limits entirely, which is how the mod behaved before 0.15.0.

   Anything under roughly $250 on the shelf is never limited, so ammunition, food,
   materials and nails are unaffected. Counted per character, and a cancelled or refused
   order gives the allowance back. */
option TheCatalogue.StockLimitMultiplier
{
    type = double,
    min = 0.0, max = 20.0,
    default = 1.0,
    page = TheCatalogue,
    translation = TheCatalogue_StockLimitMultiplier,
}

/* Minimum condition an item must have to be accepted for sale, as a fraction.
   0.0 accepts anything including near-broken gear and rotten food. */
option TheCatalogue.MinConditionToSell
{
    type = double,
    min = 0.0, max = 1.0,
    default = 0.0,
    page = TheCatalogue,
    translation = TheCatalogue_MinConditionToSell,
}

/* Everything the banking half of the mod adds -- accounts, cards, cash machines, the
   computer -- switched off in one place.

   OFF returns the mod to what it was before 0.1.0-beta: a catalogue you buy and sell with
   out of the notes in your pocket. No account, no ATM entry, no computer entry, and no
   credit card in the world is touched. That last part is the reason this option exists at
   all -- a server that does not want a banking economy, or a player who hits a conflict
   with another mod that cares about CreditCard, needs a way out that is not uninstalling. */
option TheCatalogue.BankingEnabled
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_BankingEnabled,
}

/* Whether the credit cards the world spawns belong to somebody.

   OFF leaves your OWN account working -- you can still open one at a machine and use it --
   and stops the mod naming, pricing and stamping every CreditCard it finds. Separate from
   BankingEnabled because they are different objections: one is "I do not want banking",
   the other is "I do not want the mod touching an item other things use". */
option TheCatalogue.ForeignCards
{
    type = boolean,
    default = true,
    page = TheCatalogue,
    translation = TheCatalogue_ForeignCards,
}

/* Scales the balance on a card found in the world.

   The bands behind it are weighted, not flat: half of all cards are petty cash and about
   one in twenty is worth the walk. This multiplies the result, so the SHAPE of the
   distribution is kept and only the level moves. 0 makes every found card empty, which is
   a way to keep the puzzle and remove the reward. */
option TheCatalogue.ForeignBalanceMultiplier
{
    type = double,
    min = 0.0, max = 10.0,
    default = 1.0,
    page = TheCatalogue,
    translation = TheCatalogue_ForeignBalanceMultiplier,
}

/* Wrong PINs allowed before a card is refused. Burglar always gets two more than this. */
option TheCatalogue.PinTries
{
    type = integer,
    min = 1, max = 20,
    default = 3,
    page = TheCatalogue,
    translation = TheCatalogue_PinTries,
}

/* How long a card stays refused afterwards, in game hours.

   24 is "come back tomorrow", which is the unit a player thinks in. Lower it if working
   through the twenty-four arrangements a pencil impression leaves you is too slow; set it
   to 0 to remove the lockout and keep only the count. */
option TheCatalogue.PinLockoutHours
{
    type = integer,
    min = 0, max = 168,
    default = 24,
    page = TheCatalogue,
    translation = TheCatalogue_PinLockoutHours,
}
